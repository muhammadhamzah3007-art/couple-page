    <script>
        // Array of funny/cute warnings
        const warnings = [
            "dihhhhh",
            "mang yakkk",
            "parahh betttt ",
            "sumpahhh??",
            "yakali nggak",
          
        ];

        function goToStage2() {
            document.getElementById('stage1').classList.add('hidden');
            document.getElementById('stage2').classList.remove('hidden');
        }

        function goToStage3() {
            document.getElementById('stage2').classList.add('hidden');
            document.getElementById('stage3').classList.remove('hidden');
            triggerHeartAnimation();
            playBackgroundMusic();
        }

        function avoidMouse() {
            const btn = document.getElementById('noBtn');
            const randomX = Math.random() * 200 - 100;
            const randomY = Math.random() * 200 - 100;
            
            btn.style.transform = `translate(${randomX}px, ${randomY}px)`;
            btn.style.transition = 'transform 0.3s cubic-bezier(0.34, 1.56, 0.64, 1)';
            
            // Show warning text
            const randomWarning = warnings[Math.floor(Math.random() * warnings.length)];
            document.getElementById('warningText').textContent = randomWarning;
        }

        function triggerHeartAnimation() {
            // Create multiple hearts falling
            for (let i = 0; i < 15; i++) {
                setTimeout(() => {
                    const heart = document.createElement('div');
                    heart.className = 'heart-animation';
                    heart.textContent = '❤️';
                    heart.style.left = Math.random() * window.innerWidth + 'px';
                    heart.style.top = '-30px';
                    document.body.appendChild(heart);
                    
                    // Remove element after animation
                    setTimeout(() => heart.remove(), 3000);
                }, i * 100);
            }
        }

        function playBackgroundMusic() {
            // Create a simple romantic background sound using Web Audio API
            const audioContext = new (window.AudioContext || window.webkitAudioContext)();
            const oscillator = audioContext.createOscillator();
            const gainNode = audioContext.createGain();
            
            oscillator.connect(gainNode);
            gainNode.connect(audioContext.destination);
            
            // Soft romantic tones
            oscillator.frequency.value = 320; // C note
            gainNode.gain.setValueAtTime(0.05, audioContext.currentTime);
            
            oscillator.start(audioContext.currentTime);
            oscillator.stop(audioContext.currentTime + 4);
        }

        // Add bounce animation keyframes dynamically
        const style = document.createElement('style');
        style.textContent = `
            @keyframes bounce {
                0%, 100% { transform: scale(1); }
                50% { transform: scale(1.2); }
            }
        `;
        document.head.appendChild(style);

        console.log('💕 Website loaded! Enjoy your romantic moment! 💕');
    </script>
